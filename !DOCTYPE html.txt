<script>
    const video = document.getElementById('video');
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const userTextInput = document.getElementById('userText');
    const loadingOverlay = document.getElementById('loading-overlay');

    const renderWidth = 1280;
    const renderHeight = 720;
    
    let selfieSegmentation;
    let userText = "";
    let personMask = null;

    const today = new Date();
    const yy = String(today.getFullYear()).slice(-2);
    const mm = String(today.getMonth() + 1).padStart(2, '0');
    const dd = String(today.getDate()).padStart(2, '0');
    document.getElementById('dateDisplay').innerText = `'${yy} ${mm} ${dd}`;

    async function setupCamera() {
        try {
            const stream = await navigator.mediaDevices.getUserMedia({
                video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } }
            });
            video.srcObject = stream;
            return new Promise((resolve) => {
                video.onloadedmetadata = () => resolve(video);
            });
        } catch (error) {
            alert("카메라 권한을 허용해주세요!");
        }
    }

    async function initSegmentation() {
        selfieSegmentation = new SelfieSegmentation({locateFile: (file) => {
            return `https://cdn.jsdelivr.net/npm/@mediapipe/selfie_segmentation/${file}`;
        }});
        selfieSegmentation.setOptions({ modelSelection: 1 });
        selfieSegmentation.onResults(onResults);
    }

    function onResults(results) {
        if (!video.videoWidth) return;

        if (canvas.width !== renderWidth) {
            canvas.width = renderWidth;
            canvas.height = renderHeight;
        }

        ctx.drawImage(results.image, 0, 0, canvas.width, canvas.height);

        personMask = results.segmentationMask;
        userText = userTextInput.value;
        
        if (userText !== "" && personMask) {
            drawTextBehindPerson(ctx, canvas.width, canvas.height, results.image, personMask);
        }

        addSoftGrain(ctx, canvas.width, canvas.height);
    }

    function drawTextBehindPerson(context, w, h, originalImage, mask) {
        const tempCanvas = document.createElement('canvas');
        tempCanvas.width = w;
        tempCanvas.height = h;
        const tempCtx = tempCanvas.getContext('2d');
        tempCtx.drawImage(originalImage, 0, 0, w, h);
        tempCtx.globalCompositeOperation = 'destination-in'; 
        tempCtx.drawImage(mask, 0, 0, w, h);
        
        if (userText !== "") {
            const color = getToneDownColor(originalImage);
            context.fillStyle = color;
            context.font = "120px 'VintageFont'"; 
            context.textAlign = 'center';
            context.textBaseline = 'middle';
            
            context.fillText(userText, w / 2, h / 2.2); 
        }

        context.drawImage(tempCanvas, 0, 0, w, h);
    }

    function getToneDownColor(image) {
        const tempCanvas = document.createElement('canvas');
        tempCanvas.width = 50; 
        tempCanvas.height = 50;
        const tempCtx = tempCanvas.getContext('2d');
        tempCtx.drawImage(image, 0, 0, 50, 50);
        
        const imageData = tempCtx.getImageData(0, 0, 50, 50).data;
        let r = 0, g = 0, b = 0;
        for (let i = 0; i < imageData.length; i += 4) {
            r += imageData[i];
            g += imageData[i+1];
            b += imageData[i+2];
        }
        const pixelCount = imageData.length / 4;
        r = Math.floor(r / pixelCount);
        g = Math.floor(g / pixelCount);
        b = Math.floor(b / pixelCount);

        return `rgba(${Math.floor(r*0.6)}, ${Math.floor(g*0.6)}, ${Math.floor(b*0.6)}, 0.9)`;
    }

    function addSoftGrain(context, w, h) {
        for (let i = 0; i < 1500; i++) {
            const x = Math.random() * w;
            const y = Math.random() * h;
            const size = Math.random() * 1.5 + 0.5;
            const isWhite = Math.random() > 0.5;
            context.fillStyle = isWhite ? 'rgba(255,255,255,0.05)' : 'rgba(0,0,0,0.05)';
            context.fillRect(x, y, size, size);
        }
    }

    async function drawFrame() {
        if (selfieSegmentation && video.videoWidth > 0) {
            await selfieSegmentation.send({image: video});
        }
        requestAnimationFrame(drawFrame);
    }

    async function startApp() {
        loadingOverlay.style.display = 'flex';
        await setupCamera();
        video.play();
        await initSegmentation();
        loadingOverlay.style.display = 'none';
        drawFrame(); 
    }

    // 폰트가 확실히 불려온 뒤에 프로그램 시작
    document.fonts.ready.then(() => {
        startApp();
    });
